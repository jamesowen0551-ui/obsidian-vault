from docx import Document
from docx.shared import Pt, RGBColor, Cm
from docx.enum.text import WD_ALIGN_PARAGRAPH
from docx.oxml.ns import qn

FONT = "宋体"

def set_cn(run, font=FONT):
    run.font.name = font
    run._element.rPr.rFonts.set(qn('w:eastAsia'), font)

doc = Document()
style = doc.styles['Normal']
style.font.name = FONT
style.font.size = Pt(14)   # 仿公文常用小四
style.element.rPr.rFonts.set(qn('w:eastAsia'), FONT)

for s in doc.sections:
    s.top_margin = Cm(2.54); s.bottom_margin = Cm(2.54)
    s.left_margin = Cm(2.8); s.right_margin = Cm(2.8)

def title(text, size, bold=True, center=True, after=6, color=None):
    p = doc.add_paragraph()
    if center: p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    p.paragraph_format.space_after = Pt(after)
    r = p.add_run(text); r.bold = bold; r.font.size = Pt(size)
    if color: r.font.color.rgb = RGBColor(*color)
    set_cn(r)

def h1(text):
    p = doc.add_paragraph()
    p.paragraph_format.space_before = Pt(14)
    p.paragraph_format.space_after = Pt(8)
    r = p.add_run(text); r.bold = True; r.font.size = Pt(15)
    set_cn(r)

# 段落:首句加粗总起 + 其余正文。lead为加粗总起句,rest为展开。
def para(lead, rest=""):
    p = doc.add_paragraph()
    pf = p.paragraph_format
    pf.space_after = Pt(8); pf.line_spacing = 1.55
    pf.first_line_indent = Cm(0.95)
    r1 = p.add_run(lead); r1.bold = True; r1.font.size = Pt(14); set_cn(r1)
    if rest:
        r2 = p.add_run(rest); r2.font.size = Pt(14); set_cn(r2)
